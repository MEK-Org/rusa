Widget _app(DashboardStore store, Key boundaryKey) => MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: buildMeshTheme(),
      home: Scaffold(
        backgroundColor: MeshColors.bgPrimary,
        body: RepaintBoundary(
          key: boundaryKey,
          child: DashboardBody(store: store),
        ),
      ),
    );

void main() {
  setUpAll(() async {
    await loadFonts();
  });

  for (final phase in ['before', 'after']) {
    final isAfter = phase == 'after';

    testWidgets('renders synthetic dashboard $phase charter change',
        (tester) async {
      await tester.runAsync(() async {
        final ids = _seedIds();
        HttpOverrides.global = FakeImageHttpOverrides(await portraits(ids));
        addTearDown(() => HttpOverrides.global = null);

        final api = FakeApi()
          ..threadsResult = _seedThreads(isAfter: isAfter)
          ..charters['root'] = isAfter ? rootActorCharter : defaultRootCharter;

        final store = DashboardStore(api: api, stream: FakeStream());
        await store.init();
        store.clickActor('root');

        await tester.binding.setSurfaceSize(const Size(1360, 840));
        addTearDown(() => tester.binding.setSurfaceSize(null));

        final key = GlobalKey();
        await tester.pumpWidget(_app(store, key));

        // Switch from Overview to Actors tab
        await tester.tap(find.text('Actors'));
        await tester.pump();
        await settleImages(tester, portraitUrls(ids));

        expect(find.byType(OverviewTab), findsNothing);
        expect(find.byType(ActorTree), findsOneWidget);

        // Switch detail pane to Info tab to show the Charter section
        await tester.ensureVisible(find.text('Info'));
        await tester.tap(find.text('Info'));
        for (var i = 0; i < 10; i++) {
          await tester.pump(const Duration(milliseconds: 50));
        }
        expect(find.text('Charter'), findsOneWidget);

        // 1) Wide overview: shows both ActorTree on left and Info panel with Charter on right
        await captureBoundary(key, '$_outDir/charter_overview_$phase.png');

        // 2) Focused ActorTree (tree list view)
        final treeKey = GlobalKey();
        await tester.pumpWidget(
          MaterialApp(
            debugShowCheckedModeBanner: false,
            theme: buildMeshTheme(),
            home: Scaffold(
              backgroundColor: MeshColors.bgPrimary,
              body: RepaintBoundary(
                key: treeKey,
                child: Container(
                  width: 420,
                  height: 480,
                  color: MeshColors.bgPrimary,
                  child: ActorTree(store: store),
                ),
              ),
            ),
          ),
        );
        await settleImages(tester, portraitUrls(ids));
        await captureBoundary(treeKey, '$_outDir/actor_tree_preview_$phase.png');

        // 3) Focused DetailPanel (Info tab)
        final detailKey = GlobalKey();
        await tester.pumpWidget(
          MaterialApp(
            debugShowCheckedModeBanner: false,
            theme: buildMeshTheme(),
            home: Scaffold(
              backgroundColor: MeshColors.bgPrimary,
              body: RepaintBoundary(
                key: detailKey,
                child: Container(
                  width: 650,
                  height: 750,
                  color: MeshColors.bgPrimary,
                  child: DetailPanel(store: store),
                ),
              ),
            ),
          ),
        );
        await tester.tap(find.text('Info'));
        for (var i = 0; i < 10; i++) {
          await tester.pump(const Duration(milliseconds: 50));
        }
        expect(find.text('Charter'), findsOneWidget);
        await captureBoundary(
            detailKey, '$_outDir/full_charter_detail_$phase.png');

        await store.dispose();
      });
    });
  }
}
